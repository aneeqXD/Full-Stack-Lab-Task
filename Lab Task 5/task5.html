Task 5: Modular Transcript Processing & JSON

Four ES modules that parse, validate and report transcripts, with typed errors and try/catch/finally.

NameAneeq AbdullahReg no241803ClassBSCS VACourseFSWD LabLabLab 5

Approach and explanation

Modules. Student.js has a default export; gradeUtils.js has only named exports; transcriptService.js imports both kinds and aliases one (calculateGrade as letterGrade); main.js imports from all of them. package.json sets "type": "module" so Node treats the files as ES modules.

ProblemError type
Malformed JSONSyntaxError
Not an array, missing id, missing name, missing course code, marks not a numberTypeError
Marks below 0 or above 100RangeError
Duplicate student idError

The caller wraps every case in try / catch / finally; the finally line prints whether the case passed or failed.

Student.js
// Student.js - default export
import { calculateAverage, calculateGPA } from './gradeUtils.js';

export default class Student {
  constructor(id, name, courses = []) {
    this.id = id;
    this.name = name;
    this.courses = courses;
  }
  get marks() {
    return this.courses.map((c) => c.marks);
  }
  get average() {
    return calculateAverage(this.marks);
  }
  get gpa() {
    return calculateGPA(this.marks);
  }
}
gradeUtils.js
// gradeUtils.js - named exports only
const SCALE = [
  [85, 'A', 4.0], [80, 'A-', 3.67], [75, 'B+', 3.33], [70, 'B', 3.0],
  [65, 'B-', 2.67], [60, 'C+', 2.33], [50, 'C', 2.0], [0, 'F', 0],
];
const round2 = (n) => Math.round(n * 100) / 100;

export function calculateGrade(marks) {
  const [, letter] = SCALE.find(([min]) => marks >= min);
  return letter;
}

export function calculateGradePoints(marks) {
  const [, , points] = SCALE.find(([min]) => marks >= min);
  return points;
}

export function calculateAverage(values) {
  if (!Array.isArray(values) || values.length === 0) return 0;
  return round2(values.reduce((sum, v) => sum + v, 0) / values.length);
}

export function calculateGPA(marksList) {
  return calculateAverage(marksList.map((m) => calculateGradePoints(m)));
}
transcriptService.js
// transcriptService.js - default import (Student) + named import with an alias
import Student from './Student.js';
import { calculateGrade as letterGrade } from './gradeUtils.js';

export function parseTranscripts(rawJson) {
  let data;
  try {
    data = JSON.parse(rawJson);
  } catch (err) {
    throw new SyntaxError(`Malformed JSON: ${err.message}`);
  }
  if (!Array.isArray(data)) throw new TypeError('Transcript data must be an array of students');

  const seenIds = new Set();
  return data.map((entry, index) => {
    const { id, name, courses } = entry ?? {};
    if (id === undefined || id === null || id === '') throw new TypeError(`Student at index ${index} has no id`);
    if (seenIds.has(id)) throw new Error(`Duplicate student id: ${id}`);
    seenIds.add(id);
    if (typeof name !== 'string' || name.trim() === '') throw new TypeError(`Student ${id} has no name`);
    if (!Array.isArray(courses)) throw new TypeError(`Student ${id}: courses must be an array`);

    for (const { code, marks } of courses) {
      if (typeof code !== 'string' || code.trim() === '') throw new TypeError(`Student ${id}: a course code is missing`);
      if (typeof marks !== 'number' || Number.isNaN(marks)) throw new TypeError(`Student ${id}, ${code}: marks must be a number`);
      if (marks < 0 || marks > 100) throw new RangeError(`Student ${id}, ${code}: marks ${marks} must be between 0 and 100`);
    }
    return new Student(id, name.trim(), courses);
  });
}

export function buildReport(students) {
  return {
    totalStudents: students.length,
    students: students.map((s) => ({
      id: s.id,
      name: s.name,
      average: s.average,
      gpa: s.gpa,
      courses: s.courses.map((c) => ({ ...c, grade: letterGrade(c.marks) })),
    })),
  };
}

export const reportToJSON = (report) => JSON.stringify(report, null, 2);
